import { AppLayout } from "@/components/dex/AppLayout";
import { WalletProvider } from "@/wallet/WalletContext";
import { mainnet } from "viem/chains"; // Adjust based on which chain library you use (e.g., viem, wagmi)

export default function MainLayout({ children }: { children: React.ReactNode }) {
  // Use your environment variables or fallback to sensible defaults
  const rpcUrl = process.env.NEXT_PUBLIC_RPC_URL || "https://cloudflare-eth.com";

  return (
    <WalletProvider chain={mainnet} rpcUrl={rpcUrl}>
      <AppLayout>{children}</AppLayout>
    </WalletProvider>
  );
}
